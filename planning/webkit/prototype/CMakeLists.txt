cmake_minimum_required(VERSION 3.22)
project(kobalt_webkit_m0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(abi-library SHARED abi-library.cpp)
target_link_libraries(abi-library PRIVATE Threads::Threads)
set_target_properties(abi-library PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/private")
add_executable(abi-probe abi-probe.cpp)
target_link_libraries(abi-probe PRIVATE abi-library)
set_target_properties(abi-probe PROPERTIES BUILD_RPATH "$ORIGIN/private" BUILD_RPATH_USE_ORIGIN ON)
add_executable(sandbox-probe sandbox-probe.cpp)
foreach(target abi-library abi-probe sandbox-probe)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
option(KOBALT_WEBKIT_RENDER_PROBE "Build the standalone WPE 2.0 snapshot experiment" OFF)
if(KOBALT_WEBKIT_RENDER_PROBE)
    find_package(PkgConfig REQUIRED)
    set(KOBALT_WEBKIT_PREFIX "" CACHE PATH "Explicit installed WPE prefix outside a cross sysroot")
    if(KOBALT_WEBKIT_PREFIX)
        pkg_check_modules(WPE_DEPS REQUIRED IMPORTED_TARGET glib-2.0 libsoup-3.0)
        set(wpe_library "${KOBALT_WEBKIT_PREFIX}/lib/libWPEWebKit-2.0.so")
        if(NOT EXISTS "${wpe_library}")
            message(FATAL_ERROR "No WPE library at ${wpe_library}")
        endif()
        add_library(probe-wpe INTERFACE)
        target_include_directories(probe-wpe INTERFACE
            "${KOBALT_WEBKIT_PREFIX}/include/wpe-webkit-2.0"
            "${KOBALT_WEBKIT_PREFIX}/include/wpe-webkit-2.0/wpe-platform")
        target_link_libraries(probe-wpe INTERFACE "${wpe_library}" PkgConfig::WPE_DEPS)
        set(wpe_target probe-wpe)
    else()
        pkg_check_modules(WPE REQUIRED IMPORTED_TARGET wpe-webkit-2.0 wpe-platform-2.0)
        set(wpe_target PkgConfig::WPE)
    endif()
    find_package(PNG REQUIRED)
    add_executable(wpe-snapshot snapshot.cpp)
    target_link_libraries(wpe-snapshot PRIVATE ${wpe_target} PNG::PNG)
    target_compile_options(wpe-snapshot PRIVATE -Wall -Wextra -Werror)
    set_target_properties(wpe-snapshot PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE
        INSTALL_RPATH "$ORIGIN/../private")
    configure_file(fixture.xhtml fixture.xhtml COPYONLY)
    configure_file(illustration.svg illustration.svg COPYONLY)
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../../../assets/fonts/InstrumentSerif-Regular.ttf" font.ttf COPYONLY)
endif()
