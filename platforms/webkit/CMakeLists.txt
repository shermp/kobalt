cmake_minimum_required(VERSION 3.25)
project(kobalt_webkit LANGUAGES CXX)

find_package(PkgConfig REQUIRED)
pkg_check_modules(WPE REQUIRED IMPORTED_TARGET wpe-webkit-2.0=2.54.2 wpe-platform-2.0=2.54.2)

option(KOBALT_WEBKIT_EMBEDDED_GLIB "Use the FW4 WPE library's exported GLib runtime" OFF)
if(KOBALT_WEBKIT_EMBEDDED_GLIB)
    find_library(wpe_webkit NAMES WPEWebKit-2.0 PATHS ${WPE_LIBRARY_DIRS} NO_DEFAULT_PATH REQUIRED)
endif()

include(CTest)

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS retained/location.js)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/retained/location.js" source_location_js)
configure_file(retained/location-script.h.in location-script.h @ONLY)

add_executable(wpe-retained-fixture retained/view.cpp retained/page-model.cpp retained/fixture.cpp)
target_include_directories(wpe-retained-fixture PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(wpe-retained-fixture PRIVATE cxx_std_23)
target_compile_options(wpe-retained-fixture PRIVATE -Wall -Wextra -Werror)
if(KOBALT_WEBKIT_EMBEDDED_GLIB)
    target_include_directories(wpe-retained-fixture SYSTEM PRIVATE ${WPE_INCLUDE_DIRS})
    target_compile_options(wpe-retained-fixture PRIVATE ${WPE_CFLAGS_OTHER})
    target_link_options(wpe-retained-fixture PRIVATE -static-libstdc++ -static-libgcc)
    target_link_libraries(wpe-retained-fixture PRIVATE "${wpe_webkit}")
else()
    target_link_libraries(wpe-retained-fixture PRIVATE PkgConfig::WPE)
endif()

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS retained/location-checks.js)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/retained/location-checks.js" location_checks_js)
configure_file(retained/location-checks.h.in location-checks.h @ONLY)
add_executable(wpe-location-check retained/location-check.cpp retained/page-model.cpp)
target_include_directories(wpe-location-check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(wpe-location-check PRIVATE cxx_std_23)
target_compile_options(wpe-location-check PRIVATE -Wall -Wextra -Werror)
if(KOBALT_WEBKIT_EMBEDDED_GLIB)
    target_include_directories(wpe-location-check SYSTEM PRIVATE ${WPE_INCLUDE_DIRS})
    target_compile_options(wpe-location-check PRIVATE ${WPE_CFLAGS_OTHER})
    target_link_options(wpe-location-check PRIVATE -static-libstdc++ -static-libgcc)
    target_link_libraries(wpe-location-check PRIVATE "${wpe_webkit}")
else()
    target_link_libraries(wpe-location-check PRIVATE PkgConfig::WPE)
    if(BUILD_TESTING)
        add_test(NAME wpe-source-location COMMAND wpe-location-check)
    endif()
endif()

add_executable(wpe-book-renderer socket-spike/child.cpp socket-spike/book-render.cpp
    retained/view.cpp renderer-frame.cpp)
target_compile_definitions(wpe-book-renderer PRIVATE KOBALT_BOOK_WEBKIT)
target_compile_features(wpe-book-renderer PRIVATE cxx_std_23)
target_compile_options(wpe-book-renderer PRIVATE -Wall -Wextra -Werror)
target_include_directories(wpe-book-renderer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(KOBALT_WEBKIT_EMBEDDED_GLIB)
    target_include_directories(wpe-book-renderer SYSTEM PRIVATE ${WPE_INCLUDE_DIRS})
    target_compile_options(wpe-book-renderer PRIVATE ${WPE_CFLAGS_OTHER})
    target_link_options(wpe-book-renderer PRIVATE -static-libstdc++ -static-libgcc)
    target_link_libraries(wpe-book-renderer PRIVATE "${wpe_webkit}")
else()
    target_link_libraries(wpe-book-renderer PRIVATE PkgConfig::WPE)
endif()
